    let resumeBackgroundAfterRsvpAudio = false;
    let activeRsvpAudio = null;

    function updateMusicButton(){
      const playing = !backgroundMusic.paused;
      musicButton.textContent = playing ? "🔊 Music On" : "🎵 Play Music";
      musicButton.classList.toggle("is-playing", playing);
      musicButton.setAttribute("aria-pressed", String(playing));
    }

    async function startMusic(){
      try{
        backgroundMusic.volume = 0.45;
        await backgroundMusic.play();
      }catch(error){
        // Browsers may block audible autoplay until the visitor interacts.
      }
      updateMusicButton();
    }

    musicButton.addEventListener("click", async () => {
      if(backgroundMusic.paused){
        await startMusic();
      }else{
        backgroundMusic.pause();
        updateMusicButton();
      }
    });

    backgroundMusic.addEventListener("play", updateMusicButton);
    backgroundMusic.addEventListener("pause", updateMusicButton);
    backgroundMusic.addEventListener("ended", updateMusicButton);

    async function playRsvpAudio(attendance){
      const selectedAudio = attendance === "Yes" ? yesAudio : noAudio;
      const otherAudio = attendance === "Yes" ? noAudio : yesAudio;

      // Stop any previously playing RSVP response audio.
      [selectedAudio, otherAudio].forEach(audio => {
        audio.pause();
        audio.currentTime = 0;
      });

      activeRsvpAudio = selectedAudio;
      resumeBackgroundAfterRsvpAudio = true;

      // Pause the looping background music while the RSVP response plays.
      backgroundMusic.pause();
      updateMusicButton();

      const resumeBackground = () => {
        if(activeRsvpAudio !== selectedAudio) return;
        activeRsvpAudio = null;
        selectedAudio.currentTime = 0;

        if(resumeBackgroundAfterRsvpAudio){
          resumeBackgroundAfterRsvpAudio = false;
          startMusic();
        }
      };

      selectedAudio.onended = resumeBackground;
      selectedAudio.onerror = () => {
        // If the file cannot be loaded, restore the background music.
        resumeBackground();
      };

      try{
        selectedAudio.volume = 1.0;
        await selectedAudio.play();
      }catch(error){
        console.error("Unable to play RSVP audio:", error);
        resumeBackground();
      }
    }

    function burstConfetti(){
      const colors = ["#ff5b91","#54c8e8","#ffd65a","#83d9b4","#8a71db","#ff9b5e","#ff77a4","#66c9ee","#ff8fc1","#7fdff2"];

        el.style.position = "fixed";
        el.style.zIndex = "260";
        el.style.left = "50%";
        el.style.top = "47%";
        el.style.width = (6 + Math.random()*7) + "px";
        el.style.height = (9 + Math.random()*12) + "px";
        el.style.background = colors[i % colors.length];
        el.style.pointerEvents = "none";
        el.style.borderRadius = shape === "circle" ? "50%" : "2px";
        el.style.clipPath = shape === "star"
          ? "polygon(50% 0%,61% 35%,98% 35%,68% 57%,79% 100%,50% 73%,21% 100%,32% 57%,2% 35%,39% 35%)"
          : "none";
        el.style.opacity = "1";
        el.style.transform = `rotate(${Math.random()*360}deg) scale(${0.7 + Math.random()*0.8})`;
        document.body.appendChild(el);

        requestAnimationFrame(() => {
          const angle = Math.random() * Math.PI * 2;
          const distance = 260 + Math.random() * 520;
          const x = Math.cos(angle) * distance;
          const y = Math.sin(angle) * distance - 120;
          const rotation = 540 + Math.random()*1100;
          const duration = 1200 + Math.random()*850;

          el.style.transition =
            `transform ${duration}ms cubic-bezier(.18,.7,.25,1), opacity ${duration}ms ease-out`;
          el.style.transform =
            `translate(${x}px,${y}px) rotate(${rotation}deg) scale(${0.7 + Math.random()*0.8})`;
          el.style.opacity = "0";
        });

        setTimeout(() => el.remove(), 2200);
      }

      // A second smaller burst gives the click a layered "party cannon" effect.
      setTimeout(() => {
        for(let i=0;i<24;i++){
          const el = document.createElement("i");
          el.style.position = "fixed";
          el.style.zIndex = "261";
          el.style.left = (35 + Math.random()*30) + "%";
          el.style.top = (40 + Math.random()*12) + "%";
          el.style.width = "5px";
          el.style.height = "5px";
          el.style.borderRadius = "50%";

          el.style.pointerEvents = "none";
          el.style.boxShadow = "0 0 8px rgba(255,255,255,.8)";

            const x = (Math.random()-.5) * 900;
            const y = (Math.random()-.5) * 650;

            el.style.transform = `translate(${x}px,${y}px) scale(${1.5 + Math.random()*1.5})`;
            el.style.opacity = "0";
          });

          setTimeout(() => el.remove(), 900);
        }
      }, 180);
    }
